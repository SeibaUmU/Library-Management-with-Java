package objectmodelapi;

import java.io.FileWriter;
import java.io.IOException;
import java.io.StringWriter;
import java.util.List;

import entities.SpaceMission;
import entities.SpaceModule;
import jakarta.json.Json;
import jakarta.json.JsonArray;
import jakarta.json.JsonArrayBuilder;
import jakarta.json.JsonObjectBuilder;
import jakarta.json.JsonWriter;

public class JavaPSpace {

    // 2a. Dùng Object Model API để tạo tài liệu JSON
    public String generateJsonString(List<SpaceModule> modules) {
        JsonArrayBuilder rootArrayBuilder = Json.createArrayBuilder();

        for (SpaceModule module : modules) {
            // Build array cho crewMembers
            JsonArrayBuilder crewBuilder = Json.createArrayBuilder();
            for (String member : module.getCrewMembers()) {
                crewBuilder.add(member);
            }

            // Build array cho missions
            JsonArrayBuilder missionBuilder = Json.createArrayBuilder();
            for (SpaceMission mission : module.getMissions()) {
                JsonObjectBuilder mBuilder = Json.createObjectBuilder()
                        .add("missionCode", mission.getMissionCode())
                        .add("missionTitle", mission.getMissionTitle())
                        .add("priority", mission.getPriority())
                        .add("launchDate", mission.getLaunchDate())
                        .add("estimatedCost", mission.getEstimatedCost())
                        .add("completed", mission.isCompleted());
                missionBuilder.add(mBuilder);
            }

            // Build object cho SpaceModule
            JsonObjectBuilder moduleBuilder = Json.createObjectBuilder()
                    .add("stationCode", module.getStationCode())
                    .add("stationName", module.getStationName())
                    .add("orbitNumber", module.getOrbitNumber())
                    .add("crewCapacity", module.getCrewCapacity())
                    .add("energyLevel", module.getEnergyLevel())
                    .add("crewMembers", crewBuilder)
                    .add("missions", missionBuilder);

            rootArrayBuilder.add(moduleBuilder);
        }

        JsonArray jsonArray = rootArrayBuilder.build();
        
        // Trả về chuỗi JSON
        try (StringWriter stringWriter = new StringWriter();
             JsonWriter jsonWriter = Json.createWriter(stringWriter)) {
            jsonWriter.writeArray(jsonArray);
            return stringWriter.toString();
        } catch (IOException e) {
            e.printStackTrace();
            return null;
        }
    }

    // 2b. Ghi thành file json
    public void writeJsonToFile(String jsonString, String fileName) {
        try (FileWriter fileWriter = new FileWriter(fileName)) {
            fileWriter.write(jsonString);
            System.out.println("Đã ghi file thành công: " + fileName);
        } catch (IOException e) {
            System.err.println("Lỗi khi ghi file: " + e.getMessage());
        }
    }
}

package objectmodelapi;

import java.io.FileReader;
import java.util.ArrayList;
import java.util.List;

import entities.CentralController;
import entities.GreenHouse;
import jakarta.json.Json;
import jakarta.json.JsonArray;
import jakarta.json.JsonNumber;
import jakarta.json.JsonObject;
import jakarta.json.JsonReader;
import jakarta.json.JsonValue;

public class JsonCentralController {

    // 2a. Dùng Object Model API để đọc tài liệu Json và bóc tách dữ liệu
    public List<CentralController> readJsonFile(String filePath) {
        List<CentralController> controllers = new ArrayList<>();

        try (JsonReader reader = Json.createReader(new FileReader(filePath))) {
            JsonArray rootArray = reader.readArray();

            for (JsonValue jsonValue : rootArray) {
                JsonObject obj = jsonValue.asJsonObject();

                // Lấy các thuộc tính cơ bản
                String deviceSerial = obj.getString("deviceSerial");
                
                // Xử lý các object có thể rỗng (dựa theo ảnh có 1 object trống ở cuối)
                if (!obj.containsKey("firmwareVersion")) continue; 

                String firmwareVersion = obj.getString("firmwareVersion");
                int sensorCount = obj.getInt("sensorCount");
                double batteryPercent = obj.getJsonNumber("batteryPercent").doubleValue();
                boolean automaticMode = obj.getBoolean("automaticMode");
                String lastMaintenanceDate = obj.getString("lastMaintenanceDate");

                // Đọc mảng sensorPins
                List<Integer> sensorPins = new ArrayList<>();
                JsonArray pinsArray = obj.getJsonArray("sensorPins");
                for (JsonValue pinVal : pinsArray) {
                    sensorPins.add(((JsonNumber) pinVal).intValue());
                }

                // Đọc mảng greenhouses
                List<GreenHouse> greenhouses = new ArrayList<>();
                JsonArray housesArray = obj.getJsonArray("greenhouses");
                for (JsonValue houseVal : housesArray) {
                    JsonObject hObj = houseVal.asJsonObject();
                    GreenHouse gh = new GreenHouse(
                            hObj.getInt("houseNumber"),
                            hObj.getString("cropType"),
                            hObj.getJsonNumber("lengthMeter").doubleValue(),
                            hObj.getJsonNumber("lightIntensity").doubleValue(),
                            hObj.getInt("waterTankLiter"),
                            hObj.getBoolean("irrigationEnabled")
                    );
                    greenhouses.add(gh);
                }

                // Tạo đối tượng CentralController và thêm vào danh sách
                CentralController cc = new CentralController(
                        deviceSerial, firmwareVersion, sensorCount, batteryPercent,
                        automaticMode, lastMaintenanceDate, sensorPins, greenhouses
                );
                controllers.add(cc);
            }
        } catch (Exception e) {
            e.printStackTrace();
        }
        return controllers;
    }

    // 2b. Duyệt và lấy ra GreenHouse có irrigationEnabled=true của AGR-X900
    public void printActiveIrrigation(List<CentralController> list, String targetSerial) {
        System.out.println("\n--- Các nhà kính đang bật tưới tiêu (irrigationEnabled=true) của " + targetSerial + " ---");
        for (CentralController cc : list) {
            if (cc.getDeviceSerial().equals(targetSerial)) {
                for (GreenHouse gh : cc.getGreenhouses()) {
                    if (gh.isIrrigationEnabled()) {
                        System.out.println(gh);
                    }
                }
            }
        }
    }
}

===== entities/CentralController.java =====

package entities;

import java.util.List;

public class CentralController {
    private String deviceSerial;
    private String firmwareVersion;
    private int sensorCount;
    private double batteryPercent;
    private boolean automaticMode;
    private String lastMaintenanceDate;
    private List<Integer> sensorPins;
    private List<GreenHouse> greenhouses;

    public CentralController(String deviceSerial, String firmwareVersion, int sensorCount, double batteryPercent, 
                             boolean automaticMode, String lastMaintenanceDate, List<Integer> sensorPins, List<GreenHouse> greenhouses) {
        this.deviceSerial = deviceSerial;
        this.firmwareVersion = firmwareVersion;
        this.sensorCount = sensorCount;
        this.batteryPercent = batteryPercent;
        this.automaticMode = automaticMode;
        this.lastMaintenanceDate = lastMaintenanceDate;
        this.sensorPins = sensorPins;
        this.greenhouses = greenhouses;
    }

    public String getDeviceSerial() { return deviceSerial; }
    public String getFirmwareVersion() { return firmwareVersion; }
    public int getSensorCount() { return sensorCount; }
    public double getBatteryPercent() { return batteryPercent; }
    public boolean isAutomaticMode() { return automaticMode; }
    public String getLastMaintenanceDate() { return lastMaintenanceDate; }
    public List<Integer> getSensorPins() { return sensorPins; }
    public List<GreenHouse> getGreenhouses() { return greenhouses; }

    @Override
    public String toString() {
        StringBuilder sb = new StringBuilder();
        sb.append("CentralController [Serial: ").append(deviceSerial)
          .append(", Firmware: ").append(firmwareVersion)
          .append(", Battery: ").append(batteryPercent).append("%]\n");
        sb.append("Sensor Pins: ").append(sensorPins).append("\n");
        for (GreenHouse gh : greenhouses) {
            sb.append(gh.toString()).append("\n");
        }
        return sb.toString();
    }
}

package entities;

public class GreenHouse {
    private int houseNumber;
    private String cropType;
    private double lengthMeter;
    private double lightIntensity;
    private int waterTankLiter;
    private boolean irrigationEnabled;

    public GreenHouse(int houseNumber, String cropType, double lengthMeter, double lightIntensity, int waterTankLiter, boolean irrigationEnabled) {
        this.houseNumber = houseNumber;
        this.cropType = cropType;
        this.lengthMeter = lengthMeter;
        this.lightIntensity = lightIntensity;
        this.waterTankLiter = waterTankLiter;
        this.irrigationEnabled = irrigationEnabled;
    }

    public int getHouseNumber() { return houseNumber; }
    public String getCropType() { return cropType; }
    public double getLengthMeter() { return lengthMeter; }
    public double getLightIntensity() { return lightIntensity; }
    public int getWaterTankLiter() { return waterTankLiter; }
    public boolean isIrrigationEnabled() { return irrigationEnabled; }

    @Override
    public String toString() {
        return String.format("\tGreenHouse [House: %d, Crop: %s, Length: %.1fm, Light: %.1f, Water: %dL, Irrigation: %b]",
                houseNumber, cropType, lengthMeter, lightIntensity, waterTankLiter, irrigationEnabled);
    }
}
